# Firebase Storage Rules for CollabNotes
# Copy these rules to your Firebase Console > Storage > Rules

rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {
    // Allow authenticated users to upload files to their groups
    match /groups/{groupId}/folders/{folderId}/notes/{noteId}/attachments/{fileName} {
      allow read, write: if request.auth != null 
        && exists(/databases/$(database)/documents/groups/$(groupId)/members/$(request.auth.uid));
    }
    
    // Allow test uploads for debugging
    match /test/{fileName} {
      allow read, write: if request.auth != null;
    }
    
    // Deny all other access
    match /{allPaths=**} {
      allow read, write: if false;
    }
  }
}
